// Flutter Flame: Mascote com Drag & Click
class InteractiveGoat extends SpriteAnimationComponent with DragCallbacks, TapCallbacks {
  late SpriteAnimation idleAnim, dragMoveAnim, grabAnim, fireAnim;

  @override
  Future<void> onLoad() async {
    final image = await images.load('bode_branco_24fps_master_grid.png');
    // Grade de 24 quadros (amountPerRow: 6)
    idleAnim = SpriteAnimation.fromFrameData(image, SpriteAnimationData.sequenced(
      amount: 24, amountPerRow: 6, stepTime: 1/24, textureSize: Vector2(256, 256),
    ));
    animation = idleAnim;
  }

  @override
  void onDragStart(DragStartEvent event) => animation = grabAnim;

  @override
  void onDragUpdate(DragUpdateEvent event) {
    position += event.localDelta;
    animation = dragMoveAnim;
  }

  @override
  void onDragEnd(DragEndEvent event) => animation = idleAnim;
}
